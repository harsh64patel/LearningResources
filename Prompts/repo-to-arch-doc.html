⚙️
What this batch file does: Collects three things into one file — (1) the full list of git-tracked files, (2) the contents of structurally important files (models, services, routes, config), and (3) a manifest of what was collected. Skips binaries, generated files, and anything in .gitignore.
repo_to_doc.bat
Save this to your repo root. Edit the CONFIGURATION section at the top.
batch · repo_to_doc.bat 
@echo off
setlocal EnableDelayedExpansion

:: ============================================================
::  CONFIGURATION — edit to match your project
:: ============================================================

:: Output file location
set OUTPUT=%USERPROFILE%\.continue\ai_context.txt

:: File types to EXCLUDE (add more with | separator)
set EXCL_EXT=.min.js .lock .png .jpg .jpeg .svg .gif .ico .map .pyc .pdf .woff .woff2 .zip .exe .dll

:: Folders to EXCLUDE (add more — use forward slash)
set EXCL_DIR=.tst/ migrations/ fixtures/ static/ assets/ __snapshots__/ dist/ build/

:: Keywords that identify IMPORTANT files (contents will be included)
:: These are files whose full content helps the AI understand architecture
set IMPORTANT=model service controller route handler repository schema config middleware auth

:: Max lines to include per file (prevents huge files overwhelming context)
set MAX_LINES=120

:: ============================================================
::  DO NOT EDIT BELOW THIS LINE
:: ============================================================

echo.
echo [repo_to_doc] Collecting repository context...
echo Output: %OUTPUT%
echo.

:: Clear output file
echo. > "%OUTPUT%"

:: ── SECTION 1: Repo identity ──────────────────────────────
echo === REPOSITORY: %CD% ===                              >> "%OUTPUT%"
echo Generated: %DATE% %TIME%                              >> "%OUTPUT%"
echo.                                                      >> "%OUTPUT%"

:: ── SECTION 2: Full file tree (names only) ────────────────
echo === ALL TRACKED FILES ===                             >> "%OUTPUT%"

set FILTER=git ls-files
for %%d in (%EXCL_DIR%)  do set FILTER=!FILTER! ^| findstr /v "%%d"
for %%e in (%EXCL_EXT%)  do set FILTER=!FILTER! ^| findstr /v "%%e"

%FILTER% >> "%OUTPUT%"
echo.        >> "%OUTPUT%"

:: Count files collected
for /f %%c in ('%FILTER% ^| find /c /v ""') do set FILE_COUNT=%%c
echo [repo_to_doc] File tree: %FILE_COUNT% files collected

:: ── SECTION 3: Important file contents ────────────────────
echo === KEY FILE CONTENTS ===                             >> "%OUTPUT%"
echo.                                                      >> "%OUTPUT%"

set CONTENT_COUNT=0

for /f "usebackq delims=" %%f in (`%FILTER%`) do (
    set FNAME=%%f
    set INCLUDE=0

    :: Check if filename matches any important keyword
    for %%k in (%IMPORTANT%) do (
        echo !FNAME! | findstr /i "%%k" >nul 2>&1
        if !errorlevel!==0 set INCLUDE=1
    )

    if !INCLUDE!==1 (
        set FPATH=!FNAME:/=\!
        echo --- FILE: %%f ---               >> "%OUTPUT%"

        :: Include up to MAX_LINES lines per file
        set LINENUM=0
        for /f "usebackq delims=" %%l in ("!FPATH!") do (
            if !LINENUM! lss %MAX_LINES% (
                echo %%l                     >> "%OUTPUT%"
                set /a LINENUM+=1
            )
        )

        if !LINENUM! geq %MAX_LINES% (
            echo [... truncated at %MAX_LINES% lines ...]  >> "%OUTPUT%"
        )

        echo.                                >> "%OUTPUT%"
        set /a CONTENT_COUNT+=1
    )
)

echo [repo_to_doc] File contents: %CONTENT_COUNT% key files included

:: ── SECTION 4: Package manifest ───────────────────────────
echo === DEPENDENCIES ===                                  >> "%OUTPUT%"

if exist "package.json" (
    echo --- package.json ---               >> "%OUTPUT%"
    type "package.json"                    >> "%OUTPUT%"
    echo.                                  >> "%OUTPUT%"
)

if exist "requirements.txt" (
    echo --- requirements.txt ---           >> "%OUTPUT%"
    type "requirements.txt"                >> "%OUTPUT%"
    echo.                                  >> "%OUTPUT%"
)

if exist "pyproject.toml" (
    echo --- pyproject.toml ---             >> "%OUTPUT%"
    type "pyproject.toml"                  >> "%OUTPUT%"
    echo.                                  >> "%OUTPUT%"
)

if exist "go.mod" (
    echo --- go.mod ---                     >> "%OUTPUT%"
    type "go.mod"                          >> "%OUTPUT%"
    echo.                                  >> "%OUTPUT%"
)

:: ── DONE ──────────────────────────────────────────────────
echo.
echo [repo_to_doc] Done!
echo   File:     %OUTPUT%
echo   Files:    %FILE_COUNT% tracked
echo   Contents: %CONTENT_COUNT% key files included
echo.
echo Next step in Continue.dev:
echo   @File .continue/ai_context.txt
echo   [paste the Super Prompt]
echo.
⚠
The IMPORTANT keywords list drives which files get their full contents included. Add your project's naming conventions — for example if your services are named *Svc.py or *Handler.go, add svc and handler to the list.
